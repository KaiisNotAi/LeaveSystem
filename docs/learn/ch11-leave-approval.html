1. 先想清楚狀態機

一張請假單會經過哪些狀態、由什麼事件推動？先畫出來再寫程式。

┌──────────────────────────────► Cancelled（學員自行取消） │ （只有還沒人簽核時可以） 學員送出 │ │ │ ▼ │ Pending ──────────┘ CurrentLevel=1 │ ├── 簽核人核准，而且還有下一關 ──► Pending（CurrentLevel + 1）──┐ │ │ │◄───────────────────────────────────────────────────────────────────┘ │ ├── 簽核人核准，而且已經是最後一關 ──► Approved （終點） │ └── 任何一關駁回 ─────────────────────────► Rejected （終點）

三個關鍵設計：

設計為什麼
單子有 Status，
每一關另外有 Decision		 整張單的狀態 vs 每一關的結果，是兩件事。 第 1 關核准了，整張單還是 Pending。
用 CurrentLevel 記「現在停在第幾關」 這樣「待辦清單」只要找 Level == CurrentLevel 的關卡就好，不用回推。
任一關駁回就直接結束 不做「退回上一關」。學員修正後重送一張新單， 歷史紀錄比較乾淨。

為什麼不寫成 if (hours <= 8) …

因為門檻會改。今天是 8 小時，明年可能改成 4 小時， 或者多加一級「超過 40 小時要送到局裡」。

寫死在程式裡的話，每次改都要改程式、重新編譯、重新部署。 做成資料表之後，行政人員自己在管理頁面改一改就生效了。

using System.Text.Json;

namespace MyLeaveSystem.Models;

public static class ApprovalRoleJson
{
    public static IReadOnlyList<string> Deserialize(string? json)
    {
        if (string.IsNullOrWhiteSpace(json)) return Array.Empty<string>();

        try
        {
            return JsonSerializer.Deserialize<List<string>>(json) ?? new List<string>();
        }
        catch (JsonException)
        {
            // 設定壞掉時回空清單，讓呼叫端顯示友善訊息而不是 500
            return Array.Empty<string>();
        }
    }
}

using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using MyLeaveSystem.Data;
using MyLeaveSystem.Models;
using MyLeaveSystem.Models.Entities;
using MyLeaveSystem.Models.Enums;
using MyLeaveSystem.Models.ViewModels;
using MyLeaveSystem.Services;

namespace MyLeaveSystem.Controllers;

[Authorize(Roles = "Student")]      // 整個 Controller 只有學員能進
public class LeaveRequestsController : AppControllerBase
{
    private readonly AppDbContext _db;
    private readonly ILeaveCalculator _calculator;

    public LeaveRequestsController(AppDbContext db, ILeaveCalculator calculator)
    {
        _db = db;
        _calculator = calculator;
    }

    // ── 我的請假單 ──
    public async Task<IActionResult> Index()
    {
        var studentId = GetCurrentUserId();
        if (studentId is null) return Forbid();

        var list = await _db.LeaveRequests
            .AsNoTracking()
            .Include(r => r.LeaveType)
            .Where(r => r.StudentId == studentId.Value)     // ← 只看自己的
            .OrderByDescending(r => r.CreatedAt)
            .ToListAsync();

        return View(list);
    }

    [HttpGet]
    public async Task<IActionResult> Create()
    {
        var vm = new LeaveRequestCreateViewModel();
        await PopulateLeaveTypesAsync(vm);
        return View(vm);
    }

    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Create(LeaveRequestCreateViewModel input)
    {
        // ══ 規則 0：欄位驗證 ══
        if (!ModelState.IsValid)
        {
            await PopulateLeaveTypesAsync(input);
            return View(input);
        }

        var studentId = GetCurrentUserId();
        if (studentId is null) return Forbid();

        // 連班期一起載入 —— 規則 2 要用它的上限
        var student = await _db.UserAccounts
            .Include(u => u.Cohort)
            .FirstOrDefaultAsync(u => u.Id == studentId.Value && u.IsActive);

        if (student is null) return Forbid();

        // ══ 規則 1：必須有班期 ══
        if (student.Cohort is null)
        {
            ModelState.AddModelError(string.Empty,
                "你的帳號尚未指派班期，暫時無法送出請假申請。請聯絡行政人員。");
            await PopulateLeaveTypesAsync(input);
            return View(input);
        }

        // ══ 規則 2：時數計算（第 9 章寫的 Service）══
        var startAt = input.StartDate.Date.AddHours(input.StartHour!.Value);
        var endAt   = input.EndDate.Date.AddHours(input.EndHour!.Value);

        if (!_calculator.TryCalculate(startAt, endAt, out var totalHours, out var calcError))
        {
            ModelState.AddModelError(string.Empty, calcError!);
            await PopulateLeaveTypesAsync(input);
            return View(input);
        }

        // ══ 規則 3：班期請假上限 ══
        //    口徑：只累計已核准的。Pending 不佔額度。
        var approvedHours = await _db.LeaveRequests
            .Where(r => r.StudentId == student.Id && r.Status == LeaveStatus.Approved)
            .SumAsync(r => (decimal?)r.TotalHours) ?? 0m;

        if (approvedHours + totalHours > student.Cohort.LeaveLimitHours)
        {
            ModelState.AddModelError(string.Empty,
                $"送出後將超過班期請假上限 {student.Cohort.LeaveLimitHours} 小時" +
                $"（目前已核准 {approvedHours:0.##} 小時，本次 {totalHours:0.##} 小時）。");
            await PopulateLeaveTypesAsync(input);
            return View(input);
        }

        // ══ 規則 4：依時數找出對應的簽核規則 ══
        var matchedRule = await _db.ApprovalRules
            .OrderBy(r => r.MinHours)
            .FirstOrDefaultAsync(r =>
                r.MinHours <= totalHours &&
                (!r.MaxHours.HasValue || totalHours <= r.MaxHours.Value));

        if (matchedRule is null)
        {
            ModelState.AddModelError(string.Empty, "找不到符合此請假時數的簽核規則，請聯絡行政人員。");
            await PopulateLeaveTypesAsync(input);
            return View(input);
        }

        // ══ 規則 5：依規則展開 N 關簽核步驟 ══
        var requiredRoles = ApprovalRoleJson.Deserialize(matchedRule.RequiredRoles);
        if (requiredRoles.Count == 0)
        {
            ModelState.AddModelError(string.Empty, "簽核規則未設定有效角色，請聯絡行政人員。");
            await PopulateLeaveTypesAsync(input);
            return View(input);
        }

        var steps = requiredRoles
            .Select((role, index) => new LeaveRequestStep
            {
                Level        = index + 1,      // 陣列順序 = 關卡順序
                ApproverRole = role,
                Decision     = ApprovalDecision.Pending
                // ApproverUserId 故意留 null —— 見第 6 節
            })
            .ToList();

        // ══ 全部通過，建立單子 ══
        var now = DateTime.UtcNow;
        _db.LeaveRequests.Add(new LeaveRequest
        {
            StudentId    = student.Id,
            LeaveTypeId  = input.LeaveTypeId,
            StartAt      = startAt,
            EndAt        = endAt,
            TotalHours   = totalHours,
            Reason       = input.Reason.Trim(),
            Status       = LeaveStatus.Pending,
            CurrentLevel = 1,
            CreatedAt    = now,
            UpdatedAt    = now,
            Steps        = steps          // 父子一起 Add，EF Core 會處理外鍵
        });

        await _db.SaveChangesAsync();

        TempData["Success"] = $"請假申請已送出（共 {totalHours:0.##} 小時，{steps.Count} 關簽核）。";
        return RedirectToAction(nameof(Index));
    }

    // 每次退回表單都要重新填下拉選項，不然回去是空的
    private async Task PopulateLeaveTypesAsync(LeaveRequestCreateViewModel vm)
    {
        vm.LeaveTypeOptions = await _db.LeaveTypes
            .AsNoTracking()
            .Where(t => t.IsActive)          // 停用的假別不出現在新單裡
            .OrderBy(t => t.SortOrder)
            .Select(t => new LeaveTypeOption(t.Id, t.Name))
            .ToListAsync();
    }
}
為什麼是「先全部檢查完，才建立物件」

注意所有驗證都在 _db.LeaveRequests.Add(...) 之前。 任何一關不過就直接 return View(input)， 連碰都沒碰到資料庫。

這樣就不可能出現「單子建了一半、步驟沒建成功」的殘缺資料。

每次退回都要記得重填下拉選項

PopulateLeaveTypesAsync 在這個 Action 裡出現了六次。 漏掉任何一次，使用者填錯退回表單時，假別下拉就會變成空的—— 他連改都改不了。

這是 MVC 表單最容易漏的一步，而且不會有任何錯誤訊息。

5. 待簽核清單

簽核人要看到「輪到我簽的單」。這個查詢比想像中麻煩， 因為 ApproverUserId 在還沒簽之前是 null—— 不能用 WHERE ApproverUserId = 我。

正確的判斷條件有四個，缺一不可：

#條件為什麼
1step.Decision == Pending這一關還沒簽過
2request.Status == Pending整張單還在跑（不是已駁回/取消）
3step.Level == request.CurrentLevel 只有「當前這一關」算待辦。第 3 關的人不該提前看到
4我是這個班期該角色的簽核人 光是「我是導師」不夠，還要是「這位學員班期的導師」
新增檔案Services/ApprovalService.cs（待辦清單部分）
public async Task<IReadOnlyList<LeaveRequestStep>> GetPendingForUserAsync(int currentUserId)
{
    // 這個人有哪些角色（重新查資料庫，不信 Cookie —— 原因見第 6 節）
    var roleNames = await _db.UserRoles
        .Include(ur => ur.Role)
        .Where(ur => ur.UserId == currentUserId)
        .Select(ur => ur.Role.Name)
        .ToListAsync();

    if (roleNames.Count == 0) return Array.Empty<LeaveRequestStep>();

    // ① 在資料庫端能篩的先篩掉（條件 1~3）
    var candidates = await _db.LeaveRequestSteps
        .Include(s => s.LeaveRequest).ThenInclude(r => r.Student).ThenInclude(u => u.Cohort)
        .Include(s => s.LeaveRequest).ThenInclude(r => r.LeaveType)
        .Where(s =>
            s.Decision == ApprovalDecision.Pending &&
            s.LeaveRequest.Status == LeaveStatus.Pending &&
            s.Level == s.LeaveRequest.CurrentLevel &&
            roleNames.Contains(s.ApproverRole))
        .AsNoTracking()
        .ToListAsync();

    // ② 條件 4 在記憶體端做（switch 運算式翻不成 SQL）
    return candidates
        .Where(s => GetAssignedApproverId(s) == currentUserId)
        .OrderByDescending(s => s.LeaveRequest.CreatedAt)
        .ToList();
}

/// 依關卡角色，查出這個班期指定的簽核人是誰。
private static int? GetAssignedApproverId(LeaveRequestStep step)
{
    var cohort = step.LeaveRequest.Student.Cohort;
    if (cohort is null) return null;

    return step.ApproverRole switch
    {
        "Tutor"          => cohort.TutorUserId,
        "SectionChief"   => cohort.SectionChiefUserId,
        "BranchDirector" => cohort.BranchDirectorUserId,
        _ => null
    };
}

6. 簽核引擎與五道防線

核准這個動作，驗證的程式碼比實際做事的程式碼還多。這是對的。

C#Services/ApprovalService.cs（核准部分）
public record ApprovalActionResult(bool Success, string? ErrorMessage, bool FullyApproved);

public async Task<ApprovalActionResult> ApproveAsync(int stepId, int currentUserId, string? comment)
{
    // 一次把三層關聯載齊（Step → Request → Student → Cohort）
    var step = await _db.LeaveRequestSteps
        .Include(s => s.LeaveRequest).ThenInclude(r => r.Student).ThenInclude(u => u.Cohort)
        .FirstOrDefaultAsync(s => s.Id == stepId);

    if (step is null)
        return new(false, "找不到指定的簽核關卡。", false);

    var check = await EnsureCanActAsync(step, currentUserId);
    if (!check.Success) return check;

    var now = DateTime.UtcNow;
    var request = step.LeaveRequest;

    // 寫下這一關的結果（是「寫上去」不是「改掉」，歷史要留著）
    step.Decision       = ApprovalDecision.Approved;
    step.ApproverUserId = currentUserId;       // ← 這時候才知道實際是誰簽的
    step.Comment        = string.IsNullOrWhiteSpace(comment) ? null : comment.Trim();
    step.DecidedAt      = now;

    // 還有沒有下一關？
    var nextStep = await _db.LeaveRequestSteps
        .Where(s => s.LeaveRequestId == request.Id && s.Level > request.CurrentLevel)
        .OrderBy(s => s.Level)
        .FirstOrDefaultAsync();

    bool fullyApproved;
    if (nextStep is not null)
    {
        request.CurrentLevel = nextStep.Level;      // 轉下一關，Status 仍是 Pending
        fullyApproved = false;
    }
    else
    {
        request.Status = LeaveStatus.Approved;      // 最後一關 → 整張核准
        fullyApproved = true;
    }

    request.UpdatedAt = now;
    await _db.SaveChangesAsync();

    return new(true, null, fullyApproved);
}

6.1 五道防線

C#Services/ApprovalService.cs（權限檢查）
private async Task<ApprovalActionResult> EnsureCanActAsync(LeaveRequestStep step, int currentUserId)
{
    var request = step.LeaveRequest;

    // ① 整張單必須還在跑
    if (request.Status != LeaveStatus.Pending)
        return new(false, "這張請假單已結案，無法再簽核。", false);

    // ② 必須正好是當前這一關（擋插隊）
    if (step.Level != request.CurrentLevel)
        return new(false, "這一關尚未輪到，或已經過關。", false);

    // ③ 這一關還沒被簽過（擋連點兩下、擋重送）
    if (step.Decision != ApprovalDecision.Pending)
        return new(false, "本關已簽核。", false);

    // ④ 這個人確實有該角色 —— 重新查資料庫，不信 Cookie
    var hasRole = await _db.UserRoles
        .Include(ur => ur.Role)
        .AnyAsync(ur => ur.UserId == currentUserId && ur.Role.Name == step.ApproverRole);

    if (!hasRole)
        return new(false, "您的角色無法簽核這一關。", false);

    // ⑤ 而且是「這個班期」指定的那一位（擋 A 班導師簽 B 班的單）
    var assigned = GetAssignedApproverId(step);
    if (assigned is null || assigned.Value != currentUserId)
        return new(false, "您不是這個班期指定的簽核人。", false);

    return new(true, null, false);
}
第 ④ 道為什麼要重新查資料庫？Cookie 裡不是有角色了嗎？

因為 Cookie 裡的角色是登入當下的快照，有效期 8 小時。

假設某位導師被停權或改成一般職員，他手上那個還沒過期的 Cookie 仍然帶著「Tutor」這個角色。如果只信 Cookie，他還能繼續簽 8 小時。

原則：顯示用的可以信 Cookie（快）， 真正動資料的關鍵操作要重新查（準）。

第 ⑤ 道：這就是送單時故意不寫死簽核人的收成

第 4 節建立 Step 時，ApproverUserId 留 null， 只記角色字串。「誰」是簽核當下才從班期查出來的。

好處：班期換導師時，所有在途的單立刻轉給新導師， 不用去改任何一筆歷史資料。如果送單時就寫死，那些單會永遠卡在離職的人身上。

[Authorize(Roles = "Tutor,SectionChief,BranchDirector")]
public class ApprovalsController : AppControllerBase
{
    private readonly IApprovalService _approvalService;
    public ApprovalsController(IApprovalService s) => _approvalService = s;

    public async Task<IActionResult> Index()
    {
        var userId = GetCurrentUserId();
        if (userId is null) return Forbid();

        return View(await _approvalService.GetPendingForUserAsync(userId.Value));
    }

    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Approve(int stepId, string? comment)
    {
        var userId = GetCurrentUserId();
        if (userId is null) return Forbid();

        var result = await _approvalService.ApproveAsync(stepId, userId.Value, comment);

        if (!result.Success)
            TempData["Error"] = result.ErrorMessage;
        else
            TempData["Success"] = result.FullyApproved
                ? "簽核成功，整張請假單已核准。"
                : "簽核成功，已推進到下一關。";

        return RedirectToAction(nameof(Index));
    }
}

注意 Controller 一個業務判斷都沒有—— 它只是把結果翻譯成畫面訊息。這就是第 9 章講的分層。

using Microsoft.EntityFrameworkCore;

namespace MyLeaveSystem.Tests;

public class ApprovalServiceTests

    // 每個測試一個全新的記憶體資料庫，彼此不干擾
    private static AppDbContext NewDb()

InMemory 資料庫的限制

它不是真的 SQL Server，不會檢查外鍵、不支援原生 SQL。 所以它適合測業務邏輯（誰能簽、狀態怎麼變）， 不適合測「這個查詢翻譯成的 SQL 對不對」。

完成品專案的 106 個測試就是這樣分的：邏輯用 InMemory， SQL 相關的細節靠手動驗收。
